        <select id="sizeFilter">

            <option value="">全部尺码</option>

                </tr>

            </thead>

            <tbody id="tableBody">

                <!-- 数据将通过 JS 动态填入 -->

            </tbody>

<script>

    // 为了能在 GitHub Pages 直接运行且不需要额外文件，我们先把演示数据直接写在这里

    // 等你熟悉了，再改回从 data.json 读取

    const clothesData = [

        { brand: "Nike", name: "运动夹克", size: "M", chest: 104, length: 68, height: "170-175", weight: "65-70" },

        { brand: "Uniqlo", name: "摇粒绒外套", size: "L", chest: 108, length: 70, height: "175-180", weight: "70-75" },

        { brand: "迪卡侬", name: "冲锋衣", size: "XL", chest: 112, length: 74, height: "180-185", weight: "80-85" },

        { brand: "迪卡侬", name: "滑雪服", size: "S", chest: 96, length: 62, height: "160-165", weight: "50-55" }

    ];


    // 获取 DOM 元素

    const tableBody = document.getElementById('tableBody');

    const searchInput = document.getElementById('searchInput');

    const sizeFilter = document.getElementById('sizeFilter');

    const noResult = document.getElementById('noResult');


    // 渲染表格

    function renderTable(data) {

        tableBody.innerHTML = '';

        if (data.length === 0) {

            noResult.style.display = 'block';

            return;

        }

        noResult.style.display = 'none';


        data.forEach(item => {

            const tr = document.createElement('tr');

            tr.innerHTML = `

                <td>${item.brand || '-'}</td>

                <td>${item.name || '-'}</td>

                <td><strong>${item.size || '-'}</strong></td>

                <td>${item.chest || '-'}</td>

                <td>${item.length || '-'}</td>

                <td>${item.height || '-'}</td>

                <td>${item.weight || '-'}</td>

            `;

            tableBody.appendChild(tr);

        });

    }


    // 搜索与筛选逻辑

    function filterData() {

        const keyword = searchInput.value.toLowerCase().trim();

        const selectedSize = sizeFilter.value;


        const filtered = clothesData.filter(item => {

            const matchKeyword = (item.brand && item.brand.toLowerCase().includes(keyword)) || 

                                 (item.name && item.name.toLowerCase().includes(keyword));

            const matchSize = selectedSize === '' || item.size === selectedSize;


            return matchKeyword && matchSize;

        });


        renderTable(filtered);

    }


    // 绑定事件

    searchInput.addEventListener('input', filterData);

    sizeFilter.addEventListener('change', filterData);


    // 初始渲染

    renderTable(clothesData);

</script>
